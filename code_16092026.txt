-- 📊 LIVE DISCORD STATS — NO BOOST NEEDED
-- ==============================================
local DISCORD_INVITE = "24HN79yPY"

local function GetDiscordStats()
    local total, online = "110", "7" -- Default fallback
    pcall(function()
        local url = "https://discord.com/api/v10/invites/"..DISCORD_INVITE.."?with_counts=true"
        local res = Http:GetAsync(url)
        local data = Http:JSONDecode(res)
        if data and data.approximate_member_count then
            total = tostring(data.approximate_member_count)
            online = tostring(data.approximate_presence_count)
        end
    end)
    return total, online
end
